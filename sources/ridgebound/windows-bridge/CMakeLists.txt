cmake_minimum_required(VERSION 3.16)
project(RidgeboundWheelBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(core_test tests/core_test.cpp)
enable_testing()
add_test(NAME bridge_core COMMAND core_test)
if(WIN32)
  add_executable(ridgebound-wheel-bridge bridge.cpp)
  target_link_libraries(ridgebound-wheel-bridge PRIVATE dinput8 dxguid ws2_32 ole32 shell32 user32)
  if(MINGW)
    target_compile_options(ridgebound-wheel-bridge PRIVATE -municode -Wall -Wextra -Wpedantic)
    target_link_options(ridgebound-wheel-bridge PRIVATE -municode -static -static-libgcc -static-libstdc++)
  elseif(MSVC)
    target_compile_options(ridgebound-wheel-bridge PRIVATE /W4 /utf-8)
    set_property(TARGET ridgebound-wheel-bridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endif()
